export default async (req) => {

    if (req.method !== "POST") {

        return new Response(
            JSON.stringify({
                error: "POST 요청만 가능합니다."
            }),
            {
                status: 405,
                headers: {
                    "Content-Type":
                        "application/json"
                }
            }
        );
    }


    try {

        const body =
            await req.json();

        const question =
            body.question;


        if (!question) {

            return new Response(
                JSON.stringify({
                    error:
                        "질문이 없습니다."
                }),
                {
                    status: 400,
                    headers: {
                        "Content-Type":
                            "application/json"
                    }
                }
            );
        }


        const response =
            await fetch(
                "https://api.openai.com/v1/responses",
                {

                    method: "POST",

                    headers: {

                        "Content-Type":
                            "application/json",

                        "Authorization":
                            `Bearer ${process.env.OPENAI_API_KEY}`
                    },

                    body: JSON.stringify({

                        model: "gpt-5",

                        instructions: `
당신은 계약 및 세무 검토를 지원하는 AI 보조 시스템입니다.

답변할 때 다음 원칙을 지키십시오.

1. 사실과 추정을 구분합니다.
2. 법령 근거가 필요한 경우 근거를 명확히 제시합니다.
3. 확인되지 않은 법령 조문을 만들어내지 않습니다.
4. 세금 계산은 계산 과정과 적용 세율을 함께 설명합니다.
5. 최종 법률·세무 판단은 담당자의 확인이 필요하다는 점을 명확히 합니다.
                        `,

                        input: question
                    })
                }
            );


        const data =
            await response.json();


        if (!response.ok) {

            throw new Error(
                data.error?.message ||
                "OpenAI API 오류"
            );
        }


        return new Response(

            JSON.stringify({

                answer:
                    data.output_text

            }),

            {

                status: 200,

                headers: {
                    "Content-Type":
                        "application/json"
                }

            }
        );


    } catch (error) {

        return new Response(

            JSON.stringify({
                error: error.message
            }),

            {

                status: 500,

                headers: {
                    "Content-Type":
                        "application/json"
                }

            }
        );
    }
};
